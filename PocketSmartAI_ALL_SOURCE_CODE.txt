async def lifespan(app: FastAPI):

# PocketSmart AI

PocketSmart AI is a FastAPI + Jinja2 GenAI application for budget-aware recommendations in three domains:

1. Home Interior Planning
2. Party Planning
3. Jewelry Planning

The project follows the documented architecture: modular routes, services, models, Gemini integration, authentication/session handling, responsive HTML/CSS/JavaScript, history, mock platform links, validation, and fallback recommendations.

## 1. Install Python

Use Python 3.11 or newer.

## 2. Open the project in VS Code

Open the `PocketSmartAI` folder.

## 3. Create a virtual environment

Windows PowerShell:

```powershell
python -m venv .venv
.venv\Scripts\Activate.ps1
```

If PowerShell blocks activation:

```powershell
Set-ExecutionPolicy -ExecutionPolicy RemoteSigned -Scope CurrentUser
```

Then activate again.

## 4. Install packages

```powershell
pip install -r requirements.txt
```

## 5. Create `.env`

Copy `.env.example` to `.env`.

For AI recommendations, add your Gemini API key:

```env
GEMINI_API_KEY=your_key_here
```

If you leave it empty, the application still works using deterministic fallback recommendations.

## 6. Run

```powershell
python main.py
```

Open:

- http://127.0.0.1:8000
- Swagger API docs: http://127.0.0.1:8000/docs

## 7. Test

```powershell
pytest -q
```

## Main API routes

- `POST /register`
- `POST /login`
- `POST /logout`
- `POST /token`
- `GET /session-info`
- `GET /session-data`
- `POST /session-data`
- `GET /history`
- `GET /recommendations-details/{history_id}`
- `POST /generate-home`
- `POST /generate-party`
- `POST /generate-jewelry`

## Important note about Gemini

The project documentation names Gemini 1.5 Flash Pro. The implementation keeps the model configurable through `GEMINI_MODEL` and defaults to `gemini-2.5-flash`, so you can change it in `.env` if your Google AI account exposes another supported model.

The app does not pretend that Amazon, Flipkart, IKEA, Swiggy, Zomato, or OYO data is live. The recommendation layer creates useful platform search links. Real product/vendor APIs can be integrated later where credentials and official APIs are available.

import sqlite3
from pathlib import Path
from app.config import settings


def get_connection():
    Path(settings.database_path).parent.mkdir(parents=True, exist_ok=True)
    conn = sqlite3.connect(settings.database_path)
    conn.row_factory = sqlite3.Row
    return conn


def init_db():
    conn = get_connection()
    conn.executescript(
        """
        CREATE TABLE IF NOT EXISTS users (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            username TEXT UNIQUE NOT NULL,
            email TEXT UNIQUE NOT NULL,
            password_hash TEXT NOT NULL,
            created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
        );

        CREATE TABLE IF NOT EXISTS recommendation_history (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER NOT NULL,
            planner_type TEXT NOT NULL,
            request_json TEXT NOT NULL,
            response_json TEXT NOT NULL,
            created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
            FOREIGN KEY(user_id) REFERENCES users(id)
        );
        """
    )
    conn.commit()
    conn.close()

from fastapi import Request, HTTPException, status
from app.database import get_connection


def get_current_user(request: Request):
    user_id = request.session.get("user_id")
    if not user_id:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Please log in first.",
        )

    conn = get_connection()
    row = conn.execute(
        "SELECT id, username, email, created_at FROM users WHERE id = ?",
        (user_id,),
    ).fetchone()
    conn.close()

    if not row:
        request.session.clear()
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="User session is invalid.",
        )

    return dict(row)

import hashlib
import hmac
import os
from datetime import datetime, timedelta, timezone
from jose import jwt
from app.config import settings


ALGORITHM = "HS256"


def hash_password(password: str) -> str:
    salt = os.urandom(16)
    digest = hashlib.pbkdf2_hmac(
        "sha256",
        password.encode("utf-8"),
        salt,
        120_000,
    )
    return f"{salt.hex()}${digest.hex()}"


def verify_password(password: str, stored_hash: str) -> bool:
    try:
        salt_hex, digest_hex = stored_hash.split("$", 1)
        salt = bytes.fromhex(salt_hex)
        expected = bytes.fromhex(digest_hex)
        actual = hashlib.pbkdf2_hmac(
            "sha256",
            password.encode("utf-8"),
            salt,
            120_000,
        )
        return hmac.compare_digest(actual, expected)
    except (ValueError, TypeError):
        return False


def create_access_token(subject: str) -> str:
    expire = datetime.now(timezone.utc) + timedelta(
        minutes=settings.jwt_expire_minutes
    )
    payload = {"sub": subject, "exp": expire}
    return jwt.encode(payload, settings.jwt_secret, algorithm=ALGORITHM)

import json
import re
from typing import Any, Optional

from google import genai
from google.genai import types

from app.config import settings


def format_budget(budget: float) -> str:
    return f"₹{budget:,.2f}"


def extract_json(text: str) -> Optional[dict[str, Any]]:
    if not text:
        return None

    cleaned = text.strip()

    # Remove common markdown code fences.
    cleaned = re.sub(r"^```(?:json)?\s*", "", cleaned, flags=re.I)
    cleaned = re.sub(r"\s*```$", "", cleaned)

    try:
        value = json.loads(cleaned)
        return value if isinstance(value, dict) else None
    except json.JSONDecodeError:
        pass

    # Try to find the first JSON object in a normal-language response.
    start = cleaned.find("{")
    end = cleaned.rfind("}")
    if start >= 0 and end > start:
        try:
            value = json.loads(cleaned[start:end + 1])
            return value if isinstance(value, dict) else None
        except json.JSONDecodeError:
            return None

    return None


def _client():
    if not settings.gemini_api_key:
        return None
    return genai.Client(api_key=settings.gemini_api_key)


def generate_recommendations(
    planner: str,
    payload: dict[str, Any],
    image_bytes: bytes | None = None,
    image_mime_type: str | None = None,
) -> dict[str, Any] | None:
    client = _client()
    if client is None:
        return None

    prompt = f"""
You are PocketSmart AI, a budget-aware recommendation assistant.

Planner: {planner}
User input:
{json.dumps(payload, indent=2, ensure_ascii=False)}

Return ONLY valid JSON with exactly this shape:
{{
  "planner": "{planner}",
  "budget": number,
  "summary": "short explanation",
  "allocations": {{"category": number}},
  "recommendations": [
    {{
      "name": "item or service",
      "category": "category",
      "estimated_price": number,
      "reason": "why it fits",
      "platform": "Amazon or Flipkart or IKEA or Swiggy or Zomato or OYO",
      "search_url": "https://example.com/search"
    }}
  ],
  "tips": ["tip 1", "tip 2"],
  "source": "gemini"
}}

Rules:
- Respect the user's total budget.
- Do not claim that a price is live or guaranteed.
- Estimated prices must be sensible and should sum approximately within the budget.
- Use platform search URLs, not fake product-detail URLs.
- Keep the recommendation list practical.
"""

    contents: list[Any] = [prompt]
    if image_bytes:
        contents.append(
            types.Part.from_bytes(
                data=image_bytes,
                mime_type=image_mime_type or "image/jpeg",
            )
        )
        contents.append(
            "Use the outfit image only to infer broad visual coordination such as color and style."
        )

    try:
        response = client.models.generate_content(
            model=settings.gemini_model,
            contents=contents,
        )
        return extract_json(getattr(response, "text", "") or "")
    except Exception:
        return None

from urllib.parse import quote_plus
from app.gemini_utils import generate_recommendations


def search_url(platform: str, query: str) -> str:
    q = quote_plus(query)

    urls = {
        "Amazon": f"https://www.amazon.in/s?k={q}",
        "Flipkart": f"https://www.flipkart.com/search?q={q}",
        "IKEA": f"https://www.ikea.com/in/en/search/?q={q}",
        "Swiggy": f"https://www.swiggy.com/search?query={q}",
        "Zomato": f"https://www.zomato.com/search?q={q}",
        "OYO": f"https://www.oyorooms.com/search?location={q}",
    }
    return urls.get(platform, f"https://www.google.com/search?q={q}")


def fallback_home(data: dict) -> dict:
    budget = data["budget"]
    categories = [
        ("Lighting", data.get("lights", 0), 0.12),
        ("Fans", data.get("fans", 0), 0.14),
        ("Dining", data.get("dining_tables", 0), 0.18),
        ("Sofa", data.get("sofas", 0), 0.28),
        ("Beds", data.get("beds", 0), 0.28),
    ]

    allocations = {}
    recommendations = []

    for category, quantity, share in categories:
        if quantity > 0:
            amount = round(budget * share, 2)
            allocations[category] = amount

            platform = "IKEA" if category in {"Dining", "Sofa", "Beds"} else "Amazon"
            query = f"{data.get('style', 'modern')} {category.lower()} home"
            recommendations.append({
                "name": f"{data.get('style', 'Modern').title()} {category} option",
                "category": category,
                "estimated_price": amount,
                "reason": f"Fallback estimate for {quantity} unit(s), keeping the overall plan budget-aware.",
                "platform": platform,
                "search_url": search_url(platform, query),
            })

    if not recommendations:
        allocations = {"General Home Decor": round(budget, 2)}
        recommendations.append({
            "name": "Budget-friendly home decor bundle",
            "category": "Decor",
            "estimated_price": round(budget, 2),
            "reason": "A starter recommendation because no item quantities were selected.",
            "platform": "Amazon",
            "search_url": search_url("Amazon", f"{data.get('style', 'modern')} home decor"),
        })

    return {
        "planner": "home",
        "budget": budget,
        "summary": "A fallback home plan was generated because Gemini was unavailable or returned an unusable response.",
        "allocations": allocations,
        "recommendations": recommendations,
        "tips": [
            "Compare final prices, delivery charges, and dimensions before buying.",
            "Keep a small reserve instead of spending the entire budget.",
        ],
        "source": "fallback",
    }

def get_recommendations(planner: str, data: dict, image_bytes=None, image_mime_type=None):
    result = generate_recommendations(
        planner,
        data,
        image_bytes=image_bytes,
        image_mime_type=image_mime_type,
    )

    if result:
        result["source"] = "gemini"
        return result

from fastapi import APIRouter, Request
from fastapi.responses import RedirectResponse
from jose import JWTError, jwt

from app.auth import create_access_token, hash_password, verify_password, ALGORITHM
from app.config import settings

from app.models import LoginRequest, RegisterRequest, TokenRequest

router = APIRouter()


@router.post("/register")
async def register(data: RegisterRequest, request: Request):
    username = data.username.strip()
    email = data.email.strip().lower()

    conn = get_connection()
    try:
        existing = conn.execute(
            "SELECT id FROM users WHERE username = ? OR email = ?",
            (username, email),
        ).fetchone()

        if existing:
            return {"success": False, "message": "Username or email already exists."}

        cursor = conn.execute(
            "INSERT INTO users (username, email, password_hash) VALUES (?, ?, ?)",
            (username, email, hash_password(data.password)),
        )
        conn.commit()
        request.session["user_id"] = cursor.lastrowid
        return {"success": True, "message": "Registration successful."}
    finally:
        conn.close()


@router.post("/login")
async def login(data: LoginRequest, request: Request):
    conn = get_connection()
    row = conn.execute(
        "SELECT * FROM users WHERE username = ?",
        (data.username.strip(),),
    ).fetchone()
    conn.close()

    if not row or not verify_password(data.password, row["password_hash"]):
        return {"success": False, "message": "Invalid username or password."}

    request.session["user_id"] = row["id"]
    return {"success": True, "message": "Login successful."}

@router.post("/token")
async def token(data: TokenRequest):
    conn = get_connection()
    row = conn.execute(
        "SELECT * FROM users WHERE username = ?",
        (data.username.strip(),),
    ).fetchone()
    conn.close()

    if not row or not verify_password(data.password, row["password_hash"]):
        return {"access_token": None, "token_type": "bearer", "message": "Invalid credentials."}

    return {
        "access_token": create_access_token(str(row["id"])),
        "token_type": "bearer",
    }


@router.get("/verify-token")
async def verify_token(token: str):
    try:
        payload = jwt.decode(
            token,
            settings.jwt_secret,
            algorithms=[ALGORITHM],
        )
        return {"valid": True, "subject": payload.get("sub")}
    except JWTError:
        return {"valid": False}

def save_history(user_id: int, planner: str, request_data: dict, result: dict):
    conn = get_connection()
    conn.execute(
        """
        INSERT INTO recommendation_history

        """,
        (
            user_id,
            planner,

    conn.commit()
    conn.close()

@router.post("/generate-jewelry")
async def generate_jewelry(
    budget: float = Form(...),
    occasion: str = Form("party"),
    style: str = Form("elegant"),
    metal: str = Form("any"),
    outfit_color: str = Form(""),
    notes: str = Form(""),
    outfit_image: UploadFile | None = File(default=None),
    user=Depends(get_current_user),
):
    data = JewelryRequest(
        budget=budget,
        occasion=occasion,
        style=style,
        metal=metal,
        outfit_color=outfit_color or None,
        notes=notes,
    )
    payload = data.model_dump()

    image_bytes = None
    image_mime_type = None

    if outfit_image and outfit_image.filename:
        image_bytes = await outfit_image.read()
        image_mime_type = outfit_image.content_type

    result = get_recommendations(
        "jewelry",
        payload,
        image_bytes=image_bytes,
        image_mime_type=image_mime_type,
    )
    save_history(user["id"], "jewelry", payload, result)
    return result

        "created_at": row["created_at"],
        "request": json.loads(row["request_json"]),
        "response": json.loads(row["response_json"]),
    }

    const response = await fetch("/history");

    const box = document.getElementById("historyList");

                <span>${escapeHtml(item.created_at)}</span>

        </div>
    `).join("");
}

async function viewDetails(id) {
    const response = await fetch("/recommendations-details/" + id);
    const data = await response.json();

    if (!data.response) return;

    document.getElementById("historyList").insertAdjacentHTML(
        "afterbegin",
        renderRecommendations(data.response)
    );
}

function escapeHtml(value) {
    return String(value ?? "")
        .replaceAll("&", "&amp;")
        .replaceAll("<", "&lt;")
        .replaceAll(">", "&gt;")
        .replaceAll('"', "&quot;")
        .replaceAll("'", "&#039;");
}

function money(value) {
    const n = Number(value || 0);
    return "₹" + n.toLocaleString("en-IN", {
        maximumFractionDigits: 2
    });
}

function renderRecommendations(data) {
    if (!data || data.detail) {
        return `<div class="result-card"><p>${escapeHtml(data?.detail || "Something went wrong.")}</p></div>`;
    }

    const allocations = Object.entries(data.allocations || {})
        .map(([key, value]) =>
            `<span class="chip">${escapeHtml(key)}: ${money(value)}</span>`
        ).join("");

    const recommendations = (data.recommendations || []).map(item => `
        <div class="recommendation">
            <h3>${escapeHtml(item.name)}</h3>
            <p class="small">${escapeHtml(item.category)} • ${escapeHtml(item.platform)}</p>
            <p class="price">${money(item.estimated_price)}</p>
            <p>${escapeHtml(item.reason)}</p>
            <a class="button secondary" target="_blank" rel="noopener"
               href="${escapeHtml(item.search_url)}">Search on ${escapeHtml(item.platform)}</a>
        </div>
    `).join("");

    const tips = (data.tips || []).map(t => `<li>${escapeHtml(t)}</li>`).join("");

    return `
        <div class="result-card">
            <span class="badge">${escapeHtml(data.source || "result").toUpperCase()}</span>
            <h2>${escapeHtml(data.summary || "Your recommendation plan")}</h2>
            <p><strong>Total budget:</strong> ${money(data.budget)}</p>

            <h3>Budget allocation</h3>
            <div>${allocations}</div>

            <h3>Recommendations</h3>
            ${recommendations}

            <h3>Tips</h3>
            <ul>${tips}</ul>
        </div>
    `;
}

def register_and_login(client):
    register = client.post(
        "/register",
        json={
            "username": "testuser",
            "email": "testuser@example.com",
            "password": "password123",
        },
    )

    if not register.json().get("success"):
        client.post(
            "/login",
            json={
                "username": "testuser",
                "password": "password123",
            },
        )
